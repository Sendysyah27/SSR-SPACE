```sql
-- SSR SPACE migration 10: invite workspace (token hash, sekali pakai)
-- Jalankan di Supabase Dashboard → SQL Editor SETELAH migration_09.

create table if not exists org_invites (
  id uuid primary key default gen_random_uuid(),
  org_id uuid not null references organizations(id) on delete cascade,
  role text not null default 'member' check (role in ('member', 'admin')),
  token_hash text not null unique,
  created_by uuid not null,
  expires_at timestamptz not null default (now() + interval '7 days'),
  max_uses int not null default 1,
  used_count int not null default 0,
  revoked boolean not null default false,
  created_at timestamptz not null default now()
);

alter table org_invites enable row level security;

-- SELECT: hanya admin/org. is_org_admin = SECURITY DEFINER (migration_08),
-- jangan subquery org_members di sini (aturan anti-recursion AGENTS.md).
drop policy if exists "admin read org_invites" on org_invites;
create policy "admin read org_invites" on org_invites
  for select using (public.is_org_admin(org_id));

drop policy if exists "admin create org_invites" on org_invites;
create policy "admin create org_invites" on org_invites
  for insert with check (
    public.is_org_admin(org_id) and created_by = auth.uid()
  );

drop policy if exists "admin update org_invites" on org_invites;
create policy "admin update org_invites" on org_invites
  for update using (public.is_org_admin(org_id));

-- Join HANYA lewat RPC. sha256 bawaan Postgres (PG11+), tanpa pgcrypto.
create or replace function public.join_org_by_invite(p_token text)
returns uuid
language plpgsql
security definer
set search_path = public
as $$
declare
  v_uid uuid := auth.uid();
  inv record;
  v_hash text;
  v_org uuid;
begin
  if v_uid is null then
    raise exception 'NOT_AUTHENTICATED';
  end if;

  v_hash := encode(sha256(convert_to(trim(both from p_token), 'UTF8')), 'hex');

  select * into inv
  from org_invites
  where token_hash = v_hash
  limit 1;

  if not found or inv.revoked then
    raise exception 'INVITE_INVALID';
  end if;
  if inv.expires_at < now() then
    raise exception 'INVITE_EXPIRED';
  end if;
  if inv.used_count >= inv.max_uses then
    raise exception 'INVITE_EXHAUSTED';
  end if;

  -- idempotent: sudah member → sukses tanpa tambah used_count
  select org_id into v_org
  from org_members
  where org_id = inv.org_id and user_id = v_uid;
  if found then
    return v_org;
  end if;

  insert into org_members(org_id, user_id, role)
  values (inv.org_id, v_uid, inv.role);

  update org_invites
  set used_count = used_count + 1
  where id = inv.id;

  return inv.org_id;
end $$;

grant execute on function public.join_org_by_invite(text) to authenticated;
```

- [ ] **Step 3: Implement `invite_token.dart`**

```dart
import 'dart:convert';
import 'dart:math';
import 'package:crypto/crypto.dart';

const _inviteAlphabet = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';

String generateInviteToken([Random? random]) {
  final r = random ?? Random.secure();
  final parts = <String>[];
  for (var g = 0; g < 6; g++) {
    final buf = StringBuffer();
    for (var i = 0; i < 4; i++) {
      buf.write(_inviteAlphabet[r.nextInt(_inviteAlphabet.length)]);
    }
    parts.add(buf.toString());
  }
  return parts.join('-');
}

String normalizeInviteToken(String raw) => raw.trim().toUpperCase();

String hashInviteToken(String raw) =>
    sha256.convert(utf8.encode(normalizeInviteToken(raw))).toString();

String maskInviteToken(String token) {
  final groups = token.split('-');
  if (groups.length < 2) return token;
  return '${groups.first}-…-${groups.last}';
}

String inviteStatusLabel({
  required DateTime expiresAt,
  required bool revoked,
  required int usedCount,
  required int maxUses,
  required DateTime now,
}) {
  if (revoked) return 'Dicabut';
  if (expiresAt.isBefore(now)) return 'Kedaluwarsa';
  if (usedCount >= maxUses) return 'Habis';
  return 'Aktif';
}

String inviteErrorMessage(String rpcMessage) {
  if (rpcMessage.contains('INVITE_EXPIRED')) return 'Kode undangan kedaluwarsa.';
  if (rpcMessage.contains('INVITE_EXHAUSTED')) return 'Kode undangan sudah habis.';
  if (rpcMessage.contains('INVITE_INVALID')) return 'Kode undangan tidak valid.';
  if (rpcMessage.contains('NOT_AUTHENTICATED')) return 'Masuk dulu ke aplikasi.';
  return 'Gagal bergabung. Coba lagi.';
}
```

Catatan: `normalizeInviteToken` uppercase dulu — pastikan di SQL (Task 1) hash memakai token **sudah dinormalisasi**. Client (Task 3) wajib kirim `normalizeInviteToken(...)` ke RPC, dan `hashInviteToken` dihitung dari hasil yang sama → hash identik.

```dart
import 'package:supabase_flutter/supabase_flutter.dart';
import 'package:ssr_space/core/saas/invite_token.dart';
import 'package:ssr_space/core/saas/supabase_config.dart';

class InviteException implements Exception {
  final String message;
  InviteException(this.message);
  InviteException.fromRpc(String rpcMessage)
      : message = inviteErrorMessage(rpcMessage);
  @override
  String toString() => message;
}

class InviteService {
  SupabaseClient get _c => SupabaseConfig.client;

  Future<({String token, String id})> createInvite({
    required String orgId,
    required String role,
  }) async {
    final token = generateInviteToken();
    final uid = _c.auth.currentUser!.id;
    final row = await _c
        .from('org_invites')
        .insert({
          'org_id': orgId,
          'role': role,
          'token_hash': hashInviteToken(token),
          'created_by': uid,
        })
        .select('id')
        .single();
    return (token: token, id: row['id'] as String);
  }

  Future<List<Map<String, dynamic>>> listInvites(String orgId) async {
    final rows = await _c
        .from('org_invites')
        .select('id, role, expires_at, used_count, max_uses, revoked, created_at')
        .eq('org_id', orgId)
        .order('created_at', ascending: false);
    return List<Map<String, dynamic>>.from(rows);
  }

  Future<void> revokeInvite(String id) async {
    await _c.from('org_invites').update({'revoked': true}).eq('id', id);
  }

  Future<String> join(String rawToken) async {
    try {
      final orgId = await _c.rpc(
        'join_org_by_invite',
        params: {'p_token': normalizeInviteToken(rawToken)},
      );
      return orgId as String;
    } on PostgrestException catch (err) {
      throw InviteException.fromRpc('${err.message} ${err.code ?? ''}');
    } catch (err) {
      throw InviteException.fromRpc('$err');
    }
  }
}
```

```dart
  if (busy) return;

    return;

    setState(() => joinErr = err.message);
  } catch (_) {
    setState(() => joinErr = 'Gagal bergabung. Periksa koneksi.');
  } finally {
    if (mounted) setState(() => busy = false);
  }
}
```

```dart
import 'dart:io';
import 'package:flutter/foundation.dart';
import 'package:flutter_foreground_task/flutter_foreground_task.dart';
import 'package:shared_preferences/shared_preferences.dart';
import 'package:ssr_space/core/saas/supabase_config.dart';

String? foregroundGate({
  required bool isAndroid,
  required bool masterOn,
  required bool hasSession,
}) {
  if (!isAndroid) return 'Bukan Android';
  if (!masterOn) return 'Pelacakan mati';
  if (!hasSession) return 'Belum masuk';
  return null;
}

@pragma('vm:entry-point')
void foregroundStartCallback() {
  FlutterForegroundTask.setTaskHandler(_PassiveTaskHandler());
}

/// Service hanya menjaga proses hidup; event notif tetap di stream plugin.
class _PassiveTaskHandler extends TaskHandler {
  @override
  Future<void> onStart(DateTime timestamp, TaskStarter starter) async {}

  @override
  void onRepeatEvent(DateTime timestamp) {}

  @override
  Future<void> onDestroy(DateTime timestamp, bool isTimeout) async {}
}

class ForegroundTrack {
  static bool _inited = false;
  static const _serviceId = 256;

  static void initOnce() {
    if (_inited || !Platform.isAndroid) return;
    FlutterForegroundTask.initCommunicationPort();
    FlutterForegroundTask.init(
      androidNotificationOptions: AndroidNotificationOptions(
        channelId: 'ssr_track',
        channelName: 'Pelacakan transaksi',
        channelDescription:
            'Notifikasi saat SSR SPACE memantau transaksi masuk.',
        channelImportance: NotificationChannelImportance.LOW,
        onlyAlertOnce: true,
        showWhen: false,
      ),
      iosNotificationOptions: const IOSNotificationOptions(
        showNotification: false,
      ),
      foregroundTaskOptions: ForegroundTaskOptions(
        eventAction: ForegroundTaskEventAction.nothing(),
        autoRunOnBoot: false,
        autoRunOnMyPackageReplaced: true,
        allowWakeLock: true,
      ),
    );
    _inited = true;
  }

  static Future<void> syncFromPrefs() async {
    final prefs = await SharedPreferences.getInstance();
    final gate = foregroundGate(
      isAndroid: Platform.isAndroid,
      masterOn: prefs.getBool('track_master') ?? false,
      hasSession: SupabaseConfig.client.auth.currentSession != null,
    );
    if (gate == null) {
      await start();
    } else {
      debugPrint('foreground skip: $gate');
      await stop();
    }
  }

  static Future<void> start() async {
    if (!Platform.isAndroid) return;
    initOnce();
    if (await FlutterForegroundTask.isRunningService) return;
    final perm = await FlutterForegroundTask.checkNotificationPermission();
    if (perm != NotificationPermission.granted) {
      await FlutterForegroundTask.requestNotificationPermission();
    }
    final res = await FlutterForegroundTask.startService(
      serviceId: _serviceId,
      serviceTypes: const [ForegroundServiceTypes.specialUse],
      notificationTitle: 'SSR SPACE aktif',
      notificationText: 'Memantau transaksi masuk',
      callback: foregroundStartCallback,
    );
    if (res is ServiceRequestFailure) {
      debugPrint('foreground start gagal: ${res.error}');
    }
  }

  static Future<void> stop() async {
    if (!Platform.isAndroid) return;
    if (await FlutterForegroundTask.isRunningService) {
      await FlutterForegroundTask.stopService();
    }
  }

  static Future<bool> get isRunning async {
    if (!Platform.isAndroid) return false;
    return FlutterForegroundTask.isRunningService;
  }
}
```
